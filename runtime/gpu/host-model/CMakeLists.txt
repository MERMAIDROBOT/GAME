cmake_minimum_required(VERSION 3.20)
project(ps2gs_coverage CXX)
add_library(ps2gs_model STATIC src/gs.cpp src/gs_draw.cpp src/gs_memory.cpp)
target_include_directories(ps2gs_model PUBLIC include)
target_compile_features(ps2gs_model PUBLIC cxx_std_17)
add_executable(ps2gs_coverage_tests ../../../tests/gs_touch.cpp)
target_link_libraries(ps2gs_coverage_tests PRIVATE ps2gs_model)
enable_testing()
add_test(NAME vram_boundary COMMAND ps2gs_coverage_tests)
add_test(NAME vram_boundary_optout COMMAND ps2gs_coverage_tests off)
set_tests_properties(vram_boundary_optout PROPERTIES ENVIRONMENT "PS2_PORT_GS_TIGHT_TOUCH=0")
